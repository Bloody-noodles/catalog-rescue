from issue_analysis import analyze_issues


DECISION_SAFE = "Safe to Publish"
DECISION_REVIEW = "Publish After Review"
DECISION_BLOCKED = "Do Not Publish Yet"
DECISION_INSUFFICIENT = "Insufficient Data"


def _count_by_severity(issues, severity):
    return sum(
        int(issue.get("count", 0))
        for issue in issues
        if issue.get("severity") == severity
    )


def _count_review_items(issues):
    return sum(
        int(issue.get("count", 0))
        for issue in issues
        if issue.get("requires_review", True)
    )


def _count_automatic_items(issues):
    return sum(
        int(issue.get("count", 0))
        for issue in issues
        if issue.get("auto_fix", False)
    )


def determine_business_decision(results):
    """
    Converts technical quality results into a business-facing
    publication decision.

    This layer does not modify the catalog.
    It only interprets the existing quality results.
    """

    row_count = int(results.get("row_count", 0))
    health_score = int(results.get("health_score", 0))

    if row_count <= 0:
        return DECISION_INSUFFICIENT

    issues = analyze_issues(results)

    critical = _count_by_severity(issues, "Critical")
    high = _count_by_severity(issues, "High")
    review_items = _count_review_items(issues)

    # Critical issues always block publication.
    if critical > 0:
        return DECISION_BLOCKED

    # A very low health score or significant high-severity risk
    # means the catalog should not be published yet.
    if health_score < 60 or high >= max(5, row_count // 3):
        return DECISION_BLOCKED

    # Remaining review items mean publication should happen
    # only after those issues are checked.
    if review_items > 0:
        return DECISION_REVIEW

    return DECISION_SAFE


def get_decision_title(decision):
    titles = {
        DECISION_SAFE: "Safe to Publish",
        DECISION_REVIEW: "Publish After Review",
        DECISION_BLOCKED: "Do Not Publish Yet",
        DECISION_INSUFFICIENT: "Insufficient Data",
    }
    return titles.get(decision, "Review Required")


def get_decision_explanation(decision):
    explanations = {
        DECISION_SAFE: (
            "No unresolved quality issues currently require human review. "
            "The catalog can proceed to publication."
        ),
        DECISION_REVIEW: (
            "The catalog has unresolved issues that should be reviewed "
            "before publication."
        ),
        DECISION_BLOCKED: (
            "The catalog contains significant quality risks. "
            "Resolve or verify the flagged issues before publication."
        ),
        DECISION_INSUFFICIENT: (
            "There are not enough catalog records to make a meaningful "
            "quality or publication decision."
        ),
    }
    return explanations.get(
        decision,
        "The catalog requires further review."
    )


def get_recommended_next_action(decision):
    actions = {
        DECISION_SAFE: (
            "Proceed with your catalog upload or publishing workflow."
        ),
        DECISION_REVIEW: (
            "Review the remaining flagged records, resolve business-data "
            "issues, then run Catalog Rescue again."
        ),
        DECISION_BLOCKED: (
            "Resolve the highest-severity issues first, then re-run "
            "the analysis before publishing."
        ),
        DECISION_INSUFFICIENT: (
            "Upload a catalog containing at least one product record."
        ),
    }
    return actions.get(
        decision,
        "Review the catalog before continuing."
    )


def build_business_decision(results):
    """
    Return a complete business-facing decision payload.
    """

    decision = determine_business_decision(results)

    issues = analyze_issues(results)

    critical = _count_by_severity(issues, "Critical")
    high = _count_by_severity(issues, "High")
    medium = _count_by_severity(issues, "Medium")
    low = _count_by_severity(issues, "Low")

    review_items = _count_review_items(issues)
    automatic_items = _count_automatic_items(issues)

    row_count = int(results.get("row_count", 0))
    health_score = int(results.get("health_score", 0))

    return {
        "decision": decision,
        "title": get_decision_title(decision),
        "explanation": get_decision_explanation(decision),
        "next_action": get_recommended_next_action(decision),
        "health_score": health_score,
        "row_count": row_count,
        "issue_instances": sum(
            int(issue.get("count", 0))
            for issue in issues
        ),
        "review_items": review_items,
        "automatic_items": automatic_items,
        "critical": critical,
        "high": high,
        "medium": medium,
        "low": low,
    }

def _count_result(value):
    """
    Safely count the number of detected issue records.
    """

    if isinstance(value, dict):
        return len(value)

    if isinstance(value, list):
        return len(value)

    if isinstance(value, pd.DataFrame):
        return len(value)

    return 0

# ============================================================

def analyze_issues(results):
    """
    Convert raw quality-check results into structured issues.

    Each issue is subsequently enriched using the central
    rule registry.
    """

    issues = []

    for definition in ISSUE_DEFINITIONS:

        key = definition["key"]

        value = results.get(key, {})

        count = _count_result(value)

        # Missing values are stored as:
        #
        # {
        #     "price": 4,
        #     "category": 2
        # }
        #
        # Therefore their count must be the total number
        # of missing cells rather than the number of columns.
        if key == "missing_values":
            if isinstance(value, dict):
                count = sum(value.values())
            else:
                count = 0

        if count <= 0:
            continue

        issues.append(
            {
                "severity": definition["severity"],
                "issue": definition["issue"],
                "count": count,
                "impact": definition["impact"],
                "key": key,
            }
        )

    # --------------------------------------------------------
    # Attach centralized rule metadata.
    # --------------------------------------------------------

    return enrich_issues(issues)

def calculate_impact_summary(issues):

    summary = {
        "Critical": 0,
        "High": 0,
        "Medium": 0,
        "Low": 0,
    }

    for issue in issues:

        severity = issue.get("severity")

        count = issue.get("count", 0)

        if severity in summary:
            summary[severity] += int(count)

    return summary

def build_issue_dataframe(issues):

    if not issues:
        return pd.DataFrame(
            columns=[
                "Severity",
                "Category",
                "Issue",
                "Count",
                "Automatic Fix",
                "Human Review",
                "Business Impact",
            ]
        )

    rows = []

    for issue in issues:

        rows.append(
            {
                "Severity": issue.get("severity", ""),
                "Category": issue.get("category", ""),
                "Issue": issue.get("issue", ""),
                "Count": issue.get("count", 0),
                "Automatic Fix": (
                    "Yes"
                    if issue.get("auto_fix", False)
                    else "No"
                ),
                "Human Review": (
                    "Yes"
                    if issue.get("requires_review", True)
                    else "No"
                ),
                "Business Impact": issue.get(
                    "impact",
                    "",
                ),
            }
        )

    return pd.DataFrame(rows)

# RECORDS AFFECTED BY AUTOMATIC CLEANING
# ==================================================

def count_records_affected(change_log):

    if not isinstance(
        change_log,
        list
    ):

        return 0

    affected_rows = set()

    for change in change_log:

        if not isinstance(
            change,
            dict
        ):

            continue

        row = change.get(
            "row"
        )

        # Header changes are not catalog records.
        if row == "HEADER":
            continue

        if row is not None:

            affected_rows.add(
                row
            )

    return len(
        affected_rows
    )


# ==================================================
# ISSUE INSTANCE COUNT
# ==================================================

def count_result_items(results):

    issues = analyze_issues(
        results
    )

    return sum(
        issue.get(
            "count",
            0
        )
        for issue in issues
    )


# ==================================================
# ISSUE COUNT
# ==================================================

def calculate_issue_count(results):

    return count_result_items(
        results
    )


# ==================================================
# HUMAN REVIEW ITEMS
# ==================================================

def calculate_review_items(results):

    """
    Human review is expressed as issue instances,
    not unique records.

    One record can legitimately require review
    for multiple independent problems.
    """

    return count_result_items(
        results
    )


# ==================================================
# AFFECTED RECORD PERCENTAGE
# ==================================================

def calculate_affected_percentage(
    affected_records,
    total_records
):

    if total_records <= 0:
        return 0.0

    return (
        affected_records
        /
        total_records
    ) * 100


# ==================================================
# SCORE CHANGE
# ==================================================

def calculate_score_change(
    before_score,
    after_score
):

    return int(
        after_score
        -
        before_score
    )


# ==================================================
# BUILD IMPACT SUMMARY
# ==================================================

def build_impact_summary(
    before_results,
    after_results,
    change_log
):

    # ----------------------------------------------
    # BASIC RECORD COUNTS
    # ----------------------------------------------

    total_records = (
        before_results.get(
            "row_count",
            0
        )
    )

    # ----------------------------------------------
    # ISSUE COUNTS
    # ----------------------------------------------

    issues_before = (
        count_result_items(
            before_results
        )
    )

    issues_after = (
        count_result_items(
            after_results
        )
    )

    issues_resolved = max(
        0,
        issues_before
        -
        issues_after
    )

    # ----------------------------------------------
    # AUTOMATIC CLEANUP
    # ----------------------------------------------

    affected_records = (
        count_records_affected(
            change_log
        )
    )

    affected_percentage = (
        calculate_affected_percentage(
            affected_records,
            total_records
        )
    )

    automatic_changes = (
        len(change_log)
        if isinstance(
            change_log,
            list
        )
        else 0
    )

    # ----------------------------------------------
    # HUMAN REVIEW
    # ----------------------------------------------

    review_items = (
        calculate_review_items(
            after_results
        )
    )

    # ----------------------------------------------
    # HEALTH SCORES
    # ----------------------------------------------

    before_score = int(
        before_results.get(
            "health_score",
            0
        )
    )

    after_score = int(
        after_results.get(
            "health_score",
            0
        )
    )

    score_change = (
        calculate_score_change(
            before_score,
            after_score
        )
    )

    # ----------------------------------------------
    # SEVERITY DISTRIBUTION
    # ----------------------------------------------

    before_issues = analyze_issues(
        before_results
    )

    after_issues = analyze_issues(
        after_results
    )

    before_impact = (
        calculate_impact_summary(
            before_issues
        )
    )

    after_impact = (
        calculate_impact_summary(
            after_issues
        )
    )

    # ----------------------------------------------
    # RETURN SUMMARY
    # ----------------------------------------------

    return {

        "total_records":
            total_records,

        "issues_before":
            issues_before,

        "issues_after":
            issues_after,

        "issues_resolved":
            issues_resolved,

        "automatic_changes":
            automatic_changes,

        "affected_records":
            affected_records,

        "affected_percentage":
            affected_percentage,

        "review_items":
            review_items,

        "before_score":
            before_score,

        "after_score":
            after_score,

        "score_change":
            score_change,

        # ------------------------------------------
        # BEFORE SEVERITY
        # ------------------------------------------

        "critical_before":
            before_impact["Critical"],

        "high_before":
            before_impact["High"],

        "medium_before":
            before_impact["Medium"],

        "low_before":
            before_impact["Low"],

        # ------------------------------------------
        # AFTER SEVERITY
        # ------------------------------------------

        "critical_after":
            after_impact["Critical"],

        "high_after":
            after_impact["High"],

        "medium_after":
            after_impact["Medium"],

        "low_after":
            after_impact["Low"],

    }

import pandas as pd

from issue_analysis import analyze_issues


def get_quality_status(score, row_count=None):
    if row_count is not None and row_count <= 0:
        return "Insufficient Data"
    if score >= 90:
        return "Excellent"
    if score >= 75:
        return "Good"
    if score >= 60:
        return "Needs Attention"
    if score >= 40:
        return "Poor"
    return "Critical"


def calculate_quality_dimensions(results):
    return {
        "Completeness": float(results.get("completeness", 0)),
        "Consistency": float(results.get("consistency", 0)),
        "Validity": float(results.get("validity", 0)),
        "Uniqueness": float(results.get("uniqueness", 0)),
    }


def get_top_problems(results, limit=6):
    problems = []

    missing_values = results.get("missing_values", {})
    if isinstance(missing_values, dict):
        count = sum(missing_values.values())
        if count > 0:
            problems.append(
                {
                    "problem": "Missing values",
                    "count": count,
                    "message": f"{count} missing values detected",
                }
            )

    problem_sources = [
        ("duplicate_skus", "Duplicate SKU records", "records affected by duplicate SKUs"),
        ("duplicate_rows", "Duplicate rows", "duplicate rows detected"),
        ("invalid_prices", "Invalid prices", "invalid price records detected"),
        ("suspicious_prices", "Suspicious prices", "prices require verification"),
        ("blank_skus", "Blank SKUs", "records have blank SKUs"),
        ("invalid_sku_formats", "Invalid SKU formats", "records have invalid SKU formats"),
        ("missing_descriptions", "Missing descriptions", "products are missing descriptions"),
        ("invalid_emails", "Invalid emails", "invalid email records detected"),
        ("invalid_urls", "Invalid URLs", "invalid website URLs detected"),
        ("invalid_phones", "Invalid phone numbers", "invalid phone records detected"),
        ("category_variations", "Category inconsistencies", "category consistency issues detected"),
        ("empty_columns", "Empty columns", "completely empty columns detected"),
    ]

    for key, title, description in problem_sources:
        value = results.get(key, {})
        if isinstance(value, dict):
            count = len(value)
        elif isinstance(value, list):
            count = len(value)
        elif isinstance(value, pd.DataFrame):
            count = len(value)
        else:
            count = 0

        if count <= 0:
            continue

        problems.append(
            {
                "problem": title,
                "count": count,
                "message": f"{count} {description}",
            }
        )

    problems.sort(key=lambda item: item["count"], reverse=True)
    return problems[:limit]


def build_recommendations(results):
    recommendations = []

    if results.get("row_count", 0) <= 0:
        return [
            "Upload a catalog containing at least one product record before assessing quality."
        ]

    missing_values = results.get("missing_values", {})
    if isinstance(missing_values, dict) and sum(missing_values.values()) > 0:
        recommendations.append(
            "Review missing product fields before publishing the catalog."
        )

    if len(results.get("duplicate_skus", [])) > 0:
        recommendations.append(
            "Review duplicate SKUs and confirm which records represent the same product."
        )

    if len(results.get("duplicate_rows", [])) > 0:
        recommendations.append(
            "Remove or consolidate exact duplicate records after verification."
        )

    if len(results.get("invalid_prices", [])) > 0:
        recommendations.append(
            "Verify invalid or non-positive prices before publishing."
        )

    if len(results.get("blank_skus", [])) > 0:
        recommendations.append(
            "Assign or verify SKUs for products that currently have none."
        )

    if len(results.get("invalid_emails", [])) > 0:
        recommendations.append(
            "Correct malformed email addresses before using the catalog operationally."
        )

    if len(results.get("invalid_urls", [])) > 0:
        recommendations.append(
            "Verify website URLs before publishing product or business links."
        )

    if len(results.get("invalid_phones", [])) > 0:
        recommendations.append(
            "Verify malformed phone numbers before customer-facing use."
        )

    if len(results.get("missing_descriptions", [])) > 0:
        recommendations.append(
            "Add product descriptions where marketplace or customer-facing listings require them."
        )

    if len(results.get("suspicious_prices", [])) > 0:
        recommendations.append(
            "Manually verify unusually low or high prices before publication."
        )

    if len(results.get("invalid_sku_formats", [])) > 0:
        recommendations.append(
            "Review SKU formatting for consistency before synchronization or marketplace upload."
        )

    if len(results.get("category_variations", {})) > 0:
        recommendations.append(
            "Standardize inconsistent category names before publishing or reporting."
        )

    if len(results.get("empty_columns", [])) > 0:
        recommendations.append(
            "Review completely empty columns and remove them if they are not required."
        )

    if not recommendations:
        recommendations.append(
            "No major remediation actions are currently required."
        )

    return recommendations

def _issue_instance_count(results):

        return sum(issue.get("count", 0) for issue in analyze_issues(results))
    except Exception:

def build_business_summary(results, impact_summary=None):
    row_count = int(results.get("row_count", 0))
    score = int(results.get("health_score", 0))
    status = results.get(
        "health_status",
        get_quality_status(score, row_count=row_count),
    )

    issues = analyze_issues(results)
    issue_instances = sum(int(issue.get("count", 0)) for issue in issues)

    severity = {"Critical": 0, "High": 0, "Medium": 0, "Low": 0}
    for issue in issues:
        level = issue.get("severity", "Medium")
        if level in severity:
            severity[level] += int(issue.get("count", 0))

    automatic_instances = sum(
        int(issue.get("count", 0))
        for issue in issues
        if issue.get("auto_fix") and not issue.get("requires_review")
    )

    if row_count <= 0:
        next_action = (
            "Provide a catalog containing at least one product record so quality can be assessed."
        )
        decision = "Not assessable yet"
    elif severity["Critical"] > 0:
        next_action = "Resolve critical structural issues before publication or synchronization."
        decision = "Block until critical issues are resolved"
    elif severity["High"] > 0:
        next_action = "Prioritize high-severity issues before publication or synchronization."
        decision = "Review high-risk issues before publishing"
    elif issue_instances > 0:
        next_action = "Work through the human review queue before final publication."
        decision = "Review remaining issues"
    else:
        next_action = "Catalog passes the current quality checks and is ready for normal publishing checks."
        decision = "Ready for normal publishing checks"

    summary = {
        "decision": decision,
        "next_action": next_action,
        "issue_instances": issue_instances,
        "critical_instances": severity["Critical"],
        "high_instances": severity["High"],
        "medium_instances": severity["Medium"],
        "low_instances": severity["Low"],
        "automatic_instances": automatic_instances,
        "health_score": score,
        "status": status,
        "row_count": row_count,
    }

    if isinstance(impact_summary, dict):
        summary["affected_records"] = int(impact_summary.get("affected_records", 0))
        summary["affected_percentage"] = float(
            impact_summary.get("affected_percentage", 0)
        )
        summary["issues_resolved"] = int(impact_summary.get("issues_resolved", 0))
    else:
        summary["affected_records"] = 0
        summary["affected_percentage"] = 0.0
        summary["issues_resolved"] = 0

    return summary


def build_quality_report(results, impact_summary=None):
    completeness = float(results.get("completeness", 0))
    consistency = float(results.get("consistency", 0))
    validity = float(results.get("validity", 0))
    uniqueness = float(results.get("uniqueness", 0))
    score = int(results.get("health_score", 0))
    row_count = int(results.get("row_count", 0))
    status = results.get(
        "health_status",
        get_quality_status(score, row_count=row_count),
    )

    top_problems = get_top_problems(results)
    recommendations = build_recommendations(results)
    dimensions = {
        "Completeness": completeness,
        "Consistency": consistency,
        "Validity": validity,
        "Uniqueness": uniqueness,
    }

    return {
        "health_score": score,
        "status": status,
        "data_status": results.get("data_status", "Ready for assessment"),
        "has_data": bool(results.get("has_data", row_count > 0)),
        "completeness": completeness,
        "consistency": consistency,
        "validity": validity,
        "uniqueness": uniqueness,
        "dimensions": dimensions,
        "top_problems": top_problems,
        "recommendations": recommendations,
        "business_summary": build_business_summary(results, impact_summary),
        "row_count": row_count,
        "column_count": results.get("column_count", 0),
    }


def build_report_dataframe(results):
    dimensions = calculate_quality_dimensions(results)
    rows = [
        {"Dimension": dimension, "Score": score}
        for dimension, score in dimensions.items()
    ]
    return pd.DataFrame(rows)

"""
Catalog Rescue Action Report

Builds a structured summary of what Catalog Rescue can safely
automate versus what still requires human judgment.

This module does not modify catalog data.
It only organizes remediation decisions for reporting and UI.
"""

from action_plan import (
    ACTION_AUTO_FIX,
    ACTION_REVIEW,
    ACTION_CRITICAL,
    build_action_plan,
    summarize_actions,
    count_action_types,
)


# ============================================================
# ACTION REPORT
# ============================================================

def build_action_report(issues):
    """
    Build a complete remediation report from detected issues.
    """

    plan = build_action_plan(issues)

    instance_summary = summarize_actions(plan)

    type_summary = count_action_types(plan)

    automatic_instances = instance_summary.get(
        ACTION_AUTO_FIX,
        0,
    )

    review_instances = instance_summary.get(
        ACTION_REVIEW,
        0,
    )

    critical_instances = instance_summary.get(
        ACTION_CRITICAL,
        0,
    )

    automatic_types = type_summary.get(
        ACTION_AUTO_FIX,
        0,
    )

    review_types = type_summary.get(
        ACTION_REVIEW,
        0,
    )

    critical_types = type_summary.get(
        ACTION_CRITICAL,
        0,
    )

    total_instances = (
        automatic_instances
        + review_instances
        + critical_instances
    )

    total_types = (
        automatic_types
        + review_types
        + critical_types
    )

    return {
        "plan": plan,

        # Issue-instance totals
        "total_instances": total_instances,
        "automatic_instances": automatic_instances,
        "review_instances": review_instances,
        "critical_instances": critical_instances,

        # Distinct issue-type totals
        "total_types": total_types,
        "automatic_types": automatic_types,
        "review_types": review_types,
        "critical_types": critical_types,

        # Human-readable status
        "automation_status": (
            "Safe automatic remediation available."
            if automatic_instances > 0
            else "No automatic remediation required."
        ),

        "review_status": (
            f"{review_instances} issue instances require "
            "human review."
            if review_instances > 0
            else "No human review items remain."
        ),

        "critical_status": (
            f"{critical_instances} critical issue instances "
            "require immediate attention."
            if critical_instances > 0
            else "No critical remediation actions detected."
        ),
    }

def get_automatic_actions(report):
    """
    Return issues that are approved for safe automation.
    """

    if not isinstance(report, dict):
        return []

    return [
        issue
        for issue in report.get("plan", [])
        if issue.get("action") == ACTION_AUTO_FIX
    ]


def get_review_actions(report):
    """
    Return issues that require human review.
    """

    if not isinstance(report, dict):
        return []

    return [
        issue
        for issue in report.get("plan", [])
        if issue.get("action") == ACTION_REVIEW
    ]


def get_critical_actions(report):
    """
    Return critical remediation issues.
    """

    if not isinstance(report, dict):
        return []

    return [
        issue
        for issue in report.get("plan", [])
        if issue.get("action") == ACTION_CRITICAL
    ]
